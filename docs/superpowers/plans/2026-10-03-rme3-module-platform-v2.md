# RME3 Module Platform v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. Owner preference: batch tasks into few subagents with tight prompts, no per-task reviewer agents, never start/stop the dev servers on 4010/5173 (reuse the running one, check `/health`).

**Spec:** `docs/superpowers/specs/2026-10-03-rme3-module-platform-v2-design.md`

**Goal:** Make RME3 modules look and work like native Runly modules (with or without external AI), let custom modules extend core modules through Connections with atomic saves and safe deletes, let modules evolve their schema safely, and distribute official modules from a signed catalog.

**Architecture:** Six phases, each shippable on its own. Phase 1 fixes the authoring toolchain (module CSS, contract docs, design review). Phase 2 adds high-level `@runly/ui` blocks + a module runtime context and generates golden screens. Phase 3 adds Connections (Prisma registry + index tables, FK + trigger SQL generated by Runly ORM, generic read/write services, Inventory first). Phase 4 extends the schema diff with stable field ids, renames, conversions, backfills, archive, data migrations and backups. Phase 5 improves Builder onboarding (templates, MirAI draft, external-AI prompt). Phase 6 adds the cross-module services API, domain-event outbox and the signed catalog.

**Tech Stack:** Node.js + Hono, Prisma 7 + PostgreSQL (Supabase), Runly ORM (`@runly/module-engine`), `@runly/module-compiler`, esbuild, Tailwind CSS 4 (`@tailwindcss/node`, `@tailwindcss/oxide`), React 19 + TanStack Query, `@runly/ui`, Zod, Node built-in test runner.

**Granularity note:** Phase 1 is fully detailed (code included). Phases 2–6 are decomposed into tasks with exact files, behavior, tests and commands; before starting each of those phases, expand its tasks into step-level code in this same file (a "Phase N — detailed steps" section) using the then-current code, since earlier phases change the surfaces they build on.

// apps/api/src/services/module-css-service.js
// Utilities-only Tailwind CSS for a module's components/. Module bundles are
// compiled with esbuild, so classes the app itself never uses would not exist
// in the app stylesheet (see spec 2026-10-03 §3.3). This scans the module's
// sources and builds just those utilities with the shared Runly theme.
import fs from 'node:fs/promises'
import path from 'node:path'
import { createRequire } from 'node:module'
import { fileURLToPath } from 'node:url'
import { compile } from '@tailwindcss/node'
import { Scanner } from '@tailwindcss/oxide'

const require = createRequire(import.meta.url)
const TAILWIND_BASE = path.dirname(require.resolve('tailwindcss/package.json'))
const THEME_FILE = fileURLToPath(new URL('../../../../packages/ui/src/tailwind-theme.css', import.meta.url))
const SOURCE_EXTENSIONS = new Set(['.js', '.jsx', '.mjs'])

let inputCssPromise = null
function inputCss() {
  // theme(reference): use the tokens without re-emitting :root variables the
  // app already defines; `@theme` in the shared file becomes reference too.
  inputCssPromise ??= fs.readFile(THEME_FILE, 'utf8').then((theme) => [
    '@import "tailwindcss/theme.css" theme(reference);',
    theme.replace(/@theme\s*\{/g, '@theme reference {'),
    '@import "tailwindcss/utilities.css" layer(utilities);',
  ].join('\n'))
  return inputCssPromise
}

async function collectSources(dir) {
  const out = []
  async function walk(current) {
    const entries = await fs.readdir(current, { withFileTypes: true }).catch(() => [])
    for (const entry of entries) {
      if (entry.name === 'node_modules' || entry.name.startsWith('.')) continue
      const full = path.join(current, entry.name)
      if (entry.isDirectory()) await walk(full)
      else if (SOURCE_EXTENSIONS.has(path.extname(entry.name))) {
        out.push({ content: await fs.readFile(full, 'utf8'), extension: path.extname(entry.name).slice(1) })
      }
    }
  }
  await walk(dir)
  return out
}

export async function compileModuleCss(componentsDir) {
  const sources = await collectSources(componentsDir)
  if (!sources.length) return ''
  const candidates = new Scanner({ sources: [] }).scanFiles(sources)
  if (!candidates.length) return ''
  const compiler = await compile(await inputCss(), { base: TAILWIND_BASE, onDependency: () => {} })
  return compiler.build(candidates)
}

// Cache by source hash: same sources -> same CSS for the life of the process.
export function createModuleCssCache({ computeSourceHash }) {
  const cache = new Map()
  return async function cssFor(componentsDir) {
    const hash = await computeSourceHash(componentsDir).catch(() => null)
    const key = `${componentsDir}@${hash}`
    if (!cache.has(key)) cache.set(key, compileModuleCss(componentsDir))
    return cache.get(key)
  }
}
```

// when the bundle version changes so a republished module gets fresh CSS.
function ensureModuleStylesheet(key, version) {
  const href = `${getApiUrl()}/modules/${key}/bundle.css?v=${encodeURIComponent(version)}`
  const existing = document.head.querySelector(`link[data-module-css="${CSS.escape(key)}"]`)
  if (existing?.getAttribute('href') === href) return

  if (existing) existing.replaceWith(link)

// packages/module-compiler/src/design-review.js
// Static design review for module React sources (spec 2026-10-03 §5.3). Line
// based on purpose: no parser dependency, fast, and good enough to point an AI
// at the exact line to fix. Findings never block an upload in v1.
const RULES = [
  { rule: 'native-select', severity: 'error', re: /<select[\s>]/, message: 'Usa SelectField o ComboboxField de @runly/ui en lugar de <select>.' },
  { rule: 'native-input', severity: 'error', re: /<input[\s>]/, message: 'Usa TextField, CheckboxField o DatePickerField de @runly/ui en lugar de <input>.' },
  { rule: 'native-textarea', severity: 'error', re: /<textarea[\s>]/, message: 'Usa TextareaField de @runly/ui en lugar de <textarea>.' },
  { rule: 'window-dialog', severity: 'error', re: /window\.(confirm|alert|prompt)\s*\(/, message: 'No uses window.confirm/alert/prompt: usa ConfirmDialog o Dialog de @runly/ui.' },
  { rule: 'fixed-color', severity: 'warning', re: /\b(bg|text|border)-(white|black)\b|\b(bg|text|border)-\[#[0-9a-fA-F]{3,8}\]/, message: 'Color fijo: usa los tokens del tema (bg-card, text-foreground, border-border) para modo claro y oscuro.' },
  { rule: 'emoji', severity: 'warning', re: /\p{Extended_Pictographic}/u, message: 'Runly no usa emojis en la interfaz: usa un icono de lucide-react.' },
  { rule: 'react-default-import', severity: 'error', re: /import\s+React\b/, message: "Importa hooks por nombre (import { useState } from 'react'); no uses import React." },
]

const isScreen = (file) => /\.jsx$/.test(file.path) && /export\s+default\s+function/.test(file.content)

export function reviewComponentSources(files) {
  const findings = []
  for (const file of files) {
    if (!/\.(jsx?|mjs)$/.test(file.path)) continue
    const lines = String(file.content ?? '').split(/\r?\n/)
    lines.forEach((text, index) => {
      if (/^\s*(\/\/|\*|\/\*)/.test(text)) return
      for (const rule of RULES) {
        if (rule.re.test(text)) findings.push({ file: file.path, line: index + 1, rule: rule.rule, severity: rule.severity, message: rule.message })
      }
    })
    if (isScreen(file) && !/PageHeader|ModulePage/.test(file.content)) {
      const line = lines.findIndex((text) => /export\s+default\s+function/.test(text)) + 1
      findings.push({ file: file.path, line, rule: 'no-page-header', severity: 'warning', message: 'Toda pantalla empieza con PageHeader (o ModulePage).' })
    }
  }
  return findings
}
```
